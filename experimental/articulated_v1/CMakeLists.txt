cmake_minimum_required(VERSION 3.24)
project(box3d_articulated_v1 LANGUAGES C CXX)
get_filename_component(REPO "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
add_library(box3d_articulated_v1 SHARED src/articulated_v1.cpp "${REPO}/csrc/experimental_joint_v1.cpp")
target_include_directories(box3d_articulated_v1 PUBLIC include "${REPO}/include")
target_compile_features(box3d_articulated_v1 PUBLIC cxx_std_17)
target_compile_options(box3d_articulated_v1 PRIVATE -Wall -Wextra -Werror)
add_executable(articulated_native tests/native_smoke.cpp)
target_link_libraries(articulated_native PRIVATE box3d_articulated_v1)
add_executable(articulated_c11 tests/c_header.c)
target_include_directories(articulated_c11 PRIVATE include "${REPO}/include" "${REPO}/proposals")
target_compile_features(articulated_c11 PRIVATE c_std_11)
enable_testing()
add_test(NAME synthetic_articulated_native COMMAND articulated_native)
add_test(NAME articulated_and_legacy_abi COMMAND articulated_c11)
