cmake_minimum_required(VERSION 3.24)
project(box3d_articulated_v2 LANGUAGES C CXX)
get_filename_component(REPO "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
add_library(box3d_articulated_v2 SHARED src/articulated_v2.cpp "${REPO}/experimental/articulated_v1/src/articulated_v1.cpp" "${REPO}/csrc/experimental_joint_v1.cpp")
target_include_directories(box3d_articulated_v2 PUBLIC include "${REPO}/experimental/articulated_v1/include" "${REPO}/include")
target_compile_features(box3d_articulated_v2 PUBLIC cxx_std_17)
target_compile_options(box3d_articulated_v2 PRIVATE -Wall -Wextra -Werror)
add_executable(articulated_v2_native tests/native_smoke.cpp)
target_link_libraries(articulated_v2_native PRIVATE box3d_articulated_v2)
add_executable(articulated_v2_c11 tests/c_header.c)
target_include_directories(articulated_v2_c11 PRIVATE include "${REPO}/experimental/articulated_v1/include" "${REPO}/include" "${REPO}/proposals")
target_compile_features(articulated_v2_c11 PRIVATE c_std_11)
enable_testing()
add_test(NAME synthetic_complete_solve_native COMMAND articulated_v2_native)
add_test(NAME complete_solve_and_legacy_abi COMMAND articulated_v2_c11)
