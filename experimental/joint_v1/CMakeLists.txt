cmake_minimum_required(VERSION 3.24)
project(box3d_experimental_joint_v1 LANGUAGES C CXX)
get_filename_component(REPO "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
add_library(box3d_experimental_joint_v1 SHARED "${REPO}/csrc/experimental_joint_v1.cpp")
target_include_directories(box3d_experimental_joint_v1 PUBLIC "${REPO}/include")
target_compile_features(box3d_experimental_joint_v1 PUBLIC cxx_std_17)
target_compile_options(box3d_experimental_joint_v1 PRIVATE -Wall -Wextra -Werror)
add_executable(experimental_joint_v1_host "${REPO}/examples/experimental_joint_v1_host.cpp")
target_link_libraries(experimental_joint_v1_host PRIVATE box3d_experimental_joint_v1)
enable_testing()
add_test(NAME joint_native_host COMMAND experimental_joint_v1_host)
add_executable(experimental_joint_v1_abi "${REPO}/examples/experimental_joint_v1_abi.c")
target_include_directories(experimental_joint_v1_abi PRIVATE "${REPO}/include")
target_compile_features(experimental_joint_v1_abi PRIVATE c_std_11)
add_test(NAME joint_c_abi_and_legacy_layout COMMAND experimental_joint_v1_abi)
