cmake_minimum_required(VERSION 3.16)
project(box3d_experimental_contact_v1 LANGUAGES C CXX)
add_library(box3d_experimental_contact_v1 SHARED src/contact_v1.cpp)
target_include_directories(box3d_experimental_contact_v1 PUBLIC include)
target_compile_features(box3d_experimental_contact_v1 PUBLIC cxx_std_17)
target_compile_options(box3d_experimental_contact_v1 PRIVATE -Wall -Wextra -Werror -ffp-contract=off)
enable_testing()
add_executable(contact_v1_tests tests/contact_v1_tests.cpp)
target_link_libraries(contact_v1_tests PRIVATE box3d_experimental_contact_v1)
add_test(NAME contact_v1_native_cpu COMMAND contact_v1_tests)
